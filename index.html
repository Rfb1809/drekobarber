function doGet(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet(); var apptSheet = sheet.getSheetByName("Agendamentos"); var userSheet = sheet.getSheetByName("Usuarios"); var appointments = []; var users = []; // Ler Agendamentos if (apptSheet && apptSheet.getLastRow() > 1) { var data = apptSheet.getRange(2, 1, apptSheet.getLastRow() - 1, 10).getValues(); for (var i = 0; i < data.length; i++) { if (data[i][0] || data[i][1]) { appointments.push({ id: String(data[i][0]), clientName: String(data[i][1]), clientPhone: String(data[i][2]), barber: String(data[i][3]), service: String(data[i][4]), date: formatDate(data[i][5]), time: String(data[i][6]), price: data[i][7], payment: String(data[i][8]), status: String(data[i][9] || 'Agendado') }); } } } // Ler Usuários if (userSheet && userSheet.getLastRow() > 1) { var uData = userSheet.getRange(2, 1, userSheet.getLastRow() - 1, 6).getValues(); for (var j = 0; j < uData.length; j++) { if (uData[j][0] || uData[j][1]) { users.push({ name: String(uData[j][0]), username: String(uData[j][1]), email: String(uData[j][2]), phone: String(uData[j][3]), pass: String(uData[j][4]), role: String(uData[j][5]) }); } } } var result = JSON.stringify({ appointments: appointments, users: users }); // Suporte a JSONP se houver callback if (e && e.parameter && e.parameter.callback) { return ContentService.createTextOutput(e.parameter.callback + '(' + result + ')') .setMimeType(ContentService.MimeType.JAVASCRIPT); } return ContentService.createTextOutput(result) .setMimeType(ContentService.MimeType.JSON); } function doPost(e) { var sheet = SpreadsheetApp.getActiveSpreadsheet(); var data = JSON.parse(e.postData.contents); if (data.action === "newUser") { var userSheet = sheet.getSheetByName("Usuarios") || sheet.insertSheet("Usuarios"); if (userSheet.getLastRow() === 0) { userSheet.appendRow(["Nome Completo", "Login/Nome", "E-mail", "Telefone", "Senha", "Função"]); } userSheet.appendRow([data.payload.name, data.payload.username, data.payload.email, data.payload.phone, data.payload.pass, data.payload.role]); } if (data.action === "newAppointment") { var apptSheet = sheet.getSheetByName("Agendamentos") || sheet.insertSheet("Agendamentos"); if (apptSheet.getLastRow() === 0) { apptSheet.appendRow(["ID", "Cliente", "Telefone", "Barbeiro", "Serviço", "Data", "Hora", "Valor", "Pagamento", "Status"]); } apptSheet.appendRow([data.payload.id, data.payload.clientName, data.payload.clientPhone, data.payload.barber, data.payload.service, data.payload.date, data.payload.time, data.payload.price, data.payload.payment, data.payload.status]); } return ContentService.createTextOutput(JSON.stringify({"result": "success"})).setMimeType(ContentService.MimeType.JSON); } function formatDate(dateVal) { if (!dateVal) return ""; if (dateVal instanceof Date) { var year = dateVal.getFullYear(); var month = ("0" + (dateVal.getMonth() + 1)).slice(-2); var day = ("0" + dateVal.getDate()).slice(-2); return year + "-" + month + "-" + day; } return String(dateVal).split('T')[0]; }
